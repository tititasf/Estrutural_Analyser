def dim_nums(path):
    doc=ezdxf.readfile(path); nums=set()
    for e in doc.modelspace():
        if e.dxftype()=='TEXT':
            t=e.dxf.text.strip()
            if re.fullmatch(r'\\d+(?:\\.\\d+)?', t): nums.add(float(t))
    return nums

def face_h(path):
    doc=ezdxf.readfile(path); ys=[]
    for e in doc.modelspace():
        if e.dxftype()=='LWPOLYLINE':
            for p in e.get_points('xy'): ys.append(p[1])
        elif e.dxftype()=='LINE':
            ys.extend([e.dxf.start.y,e.dxf.end.y])
    return round(max(ys)-min(ys),1) if ys else 0

def near(a,b,tol=5,rel=0.12):
    if a<=0 or b<=0: return False
    return abs(a-b)<=max(tol, max(a,b)*rel)
